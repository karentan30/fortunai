📋 基本集成模板

在所有报告页面 (report.html, report-en.html, report-kr.html 等) 的最后，在 </body> 标签前添加以下代码：

HTML - 标准集成
<!-- 反馈弹窗脚本 -->
<script src="/assets/js/feedback-modal.js"></script>

<!-- 初始化反馈系统 -->
<script>
(function() {
  // 确保 DOM 完全加载
  function initFeedback() {
    // 报告生成完成后，延迟显示反馈弹窗
    setTimeout(function() {
      FeedbackModal.show({
        readingType: 'bazi',  // 替换为实际占算类型
        lang: 'zh',
        onSuccess: function(response) {
          console.log('反馈已提交，ID:', response.feedbackId);
        }
      });
    }, 2000);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initFeedback);
  } else {
    initFeedback();
  }
})();
</script>

// 在报告完全渲染后显示反馈弹窗
(function() {
  const targetElement = document.querySelector('.report-footer');

  if (targetElement) {
    // 使用 MutationObserver 检测报告生成完成
    const observer = new MutationObserver(function(mutations) {
      setTimeout(function() {
        FeedbackModal.show({
          readingType: 'bazi',
          lang: 'zh',
          onSuccess: function() {
            // 可选：记录用户反馈行为埋点

            }
          }
        });
        observer.disconnect();
      }, 1500);
    });

    observer.observe(targetElement, { childList: true, subtree: true });
  }
})();
</script>

🎯 高级集成示例

条件显示反馈弹窗

仅在用户首次查看报告时显示：

JavaScript - 条件显示
<script>
(function() {
  function initFeedback() {
    // 检查是否已显示过反馈弹窗
    const sessionKey = 'feedback_shown_' + new Date().toISOString().slice(0, 10);
    const shown = sessionStorage.getItem(sessionKey);

    if (shown) {
      console.log('[Feedback] 今日已显示反馈弹窗');
      return;
    }

    setTimeout(function() {
      FeedbackModal.show({
        readingType: 'bazi',
        lang: 'zh',
        onSuccess: function() {
          sessionStorage.setItem(sessionKey, '1');
        }
      });
    }, 2000);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initFeedback);
  } else {
    initFeedback();
  }
})();
</script>

集成Google Analytics埋点

JavaScript - 埋点集成
<script>
(function() {
  function initFeedback() {
    setTimeout(function() {
      FeedbackModal.show({
        readingType: 'bazi',
        lang: 'zh',
        onSuccess: function(response) {
          // Google Analytics 事件
          if (window.gtag) {
            gtag('event', 'feedback_submitted', {
              feedback_id: response.feedbackId,
              reading_type: 'bazi',
              language: 'zh'
            });
          }

          // 自定义事件
          if (window.dataLayer) {
            window.dataLayer.push({
              event: 'feedback_submitted',
              feedbackId: response.feedbackId
            });
          }
        }
      });
    }, 2000);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initFeedback);
  } else {
    initFeedback();
  }
})();
</script>

针对不同占算类型的集成

JavaScript - 动态读取占算类型
<script>
(function() {
  // 根据当前页面动态确定占算类型
  function getReadingType() {
    const pathname = window.location.pathname;

    if (pathname.includes('report')) return 'bazi';
    if (pathname.includes('tarot')) return 'tarot';
    if (pathname.includes('ziwei')) return 'ziwei';
    if (pathname.includes('hehun')) return 'hehun';
    if (pathname.includes('astrology')) return 'astrology';

    return 'general';
  }

  // 根据URL路径动态确定语言
  function getLanguage() {
    if (window.location.pathname.includes('-en')) return 'en';
    if (window.location.pathname.includes('-kr')) return 'kr';
    return 'zh';
  }

  function initFeedback() {
    setTimeout(function() {
      FeedbackModal.show({
        readingType: getReadingType(),
        lang: getLanguage()
      });
    }, 2000);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initFeedback);
  } else {
    initFeedback();
  }
})();
</script>

📊 前端埋点示例

建议在反馈系统中集成以下埋点来追踪用户行为：

JavaScript - 完整埋点方案
<script src="/assets/js/feedback-modal.js"></script>
<script>
// 反馈埋点配置
const FeedbackTracking = {
  trackOpen: function(type, lang) {
    if (window.gtag) {
      gtag('event', 'feedback_modal_opened', {
        reading_type: type,
        language: lang,
        timestamp: new Date().toISOString()
      });
    }
  },

  trackRating: function(rating) {
    if (window.gtag) {
      gtag('event', 'feedback_rating_selected', {
        rating: rating,
        timestamp: new Date().toISOString()
      });
    }
  },

  trackSubmit: function(rating, category, type) {
    if (window.gtag) {
      gtag('event', 'feedback_submitted', {
        rating: rating,
        category: category,
        reading_type: type,
        timestamp: new Date().toISOString()
      });
    }
  },

  trackClose: function(reason) {
    if (window.gtag) {
      gtag('event', 'feedback_modal_closed', {
        reason: reason,
        timestamp: new Date().toISOString()
      });
    }
  }
};

// 集成埋点
function initFeedbackWithTracking() {
  const readingType = 'bazi';
  const lang = 'zh';

  FeedbackTracking.trackOpen(readingType, lang);

  FeedbackModal.show({
    readingType: readingType,
    lang: lang,
    onSuccess: function(response) {
      FeedbackTracking.trackSubmit(
        document.getElementById('feedbackRating')?.value || 0,
        document.getElementById('feedbackCategory')?.value || 'general',
        readingType
      );
    },
    onClose: function() {
      FeedbackTracking.trackClose('user_closed');
    }
  });
}

setTimeout(initFeedbackWithTracking, 2000);
</script>

❓ 常见问题

Q: 如何隐藏用户名和邮箱字段？

A: 在 feedback-modal.js 中找到 userInfoHtml，将 display:none 改为 display:flex 即可让字段可选显示。

Q: 如何修改反馈分类选项？

A: 在 feedback-modal.js 中找到 CATEGORIES 对象，添加或修改选项：

const CATEGORIES = {
  quality: '准确度',
  custom: '我的自定义分类'  // 新增
};

Q: 如何让反馈弹窗在特定条件下显示？

A: 在 onSuccess/onClose 回调中添加条件判断，或使用 sessionStorage 来控制显示频率。

Q: 反馈数据存储在哪里？

A: 存储在内存 (_M.feedbacks) 中，每次更新后通过 _persist() 落盘到 server/data.json。
