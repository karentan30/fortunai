### 1. 印度占星 Jyotish (`pages/jyotish.html`)

**业务逻辑**
- 输入：姓名 / 出生日期 / 出生时间 / 出生城市国家
- 计算链：Gregorian Date → Julian Day → Rashi（月亮星座）+ Nakshatra（27月宿）+ Lagna（上升）+ Vimshottari Dasha（大运周期）
- 免费预览：Rashi + Nakshatra + 1 段引子
- 付费（$9.90/$19.90）：Dasha 大运 / 12 宫分析 / 5 年行星预测 / 宝石建议

**技术实现**
```js
// 前端：简化计算（精准度足够用户接受）
function calculateJyotish(birthDate, birthTime, birthCity) {
  const jd = gregorianToJulian(birthDate);
  const sunPos = calculateSunPosition(jd);  // 近似，用 DeepSeek 精算
  const moonPos = calculateMoonPosition(jd);
  const rashi = Math.floor((moonPos + 23.85) / 30);  // Lahiri ayanamsa
  const nakshatra = Math.floor((moonPos % 360) / 13.33);
  return { rashi, nakshatra, jd };
}

// 后端：DeepSeek 接手全部精确解读
// 用户输入 → /api/jyotish → DeepSeek 生成报告
```

**API 端点**（加入 `server/routes/divination.js`）
```js
router.post('/jyotish', async (req, res) => {
  const { name, dob, tob, city, country, concern, lang } = req.body;
  // 前端计算基础数据
  const jyotishData = calculateJyotish(new Date(dob), tob, city);
  // DeepSeek 生成解读
  const reading = await llm.generate({
    prompt: `[Jyotish system prompt] 
      User: ${name}, born ${dob} at ${tob} in ${city}, ${country}
      Rashi: ${jyotishData.rashi}, Nakshatra: ${jyotishData.nakshatra}
      Concern: ${concern}
      Language: ${lang}
      Generate authentic Jyotish reading...`,
    temp: 0.7
  });
  // hasFullAccess() 判断免费/付费展示
  res.json({ ...reading, tier: hasFullAccess(req, ['jyotish_full']) ? 'full' : 'basic' });
});
```

### 2. 藏传命理 (`pages/tibet.html`)

**核心计算**（所有逻辑源自出生年份，最简单）
```js
function calculateTibetan(birthYear) {
  const chineseZodiacIdx = (birthYear - 1900) % 12;  // 12 生肖
  const elementIdx = (birthYear - 1900) % 5;  // 5 元素
  const mewaNum = ((birthYear - 1) % 9) + 1;  // Mewa 数字宫 1-9
  const parkhaIdx = (birthYear - 1) % 8;  // Parkha 八卦索引
  
  // 4 生命能量（计算规则源自藏传经典）
  const sog = 15 - ((birthYear % 60) >= 30 ? 1 : 0);  // Sog 生命力
  const lu = 13 + ((birthYear % 60) % 15);  // Lü 身体能量
  const wangtang = (birthYear % 60) % 16;  // Wangtang 权力
  const lungta = (birthYear % 60) >= 30 ? 'High' : 'Low';  // Lungta 风马运势
  
  return { chineseZodiac, element, mewaNum, parkhaIdx, sog, lu, wangtang, lungta };
}
```

### 3. 玛雅历 Tzolkin (`pages/maya.html`)

**纯前端计算**（260 天周期，确定性算法）
```js
function gregorianToJulianDay(year, month, day) {
  const a = Math.floor((14 - month) / 12);
  const y = year + 4800 - a;
  const m = month + 12 * a - 3;
  return day + Math.floor((153*m+2)/5) + 365*y + Math.floor(y/4) - Math.floor(y/100) + Math.floor(y/400) - 32045;
}

function getTzolkin(year, month, day) {
  const jd = gregorianToJulianDay(year, month, day);
  const kin = ((jd - 584283) % 260 + 260) % 260;  // GMT 关联数
  const daySignIdx = kin % 20;  // 0-19（20 日印）
  const tone = (kin % 13) + 1;  // 1-13（13 音调）
  
  const dayNames = ['Imix', 'Ik', 'Akbal', 'Kan', 'Chicchan', 'Cimi', 'Manik', 'Lamat', 'Muluc', 'Oc', 
                    'Chuen', 'Eb', 'Ben', 'Ix', 'Men', 'Cib', 'Caban', 'Etznab', 'Cauac', 'Ahau'];
  return { kin, daySign: dayNames[daySignIdx], tone };
}
```

**API 端点**（`server/routes/divination.js`）
```js
router.post('/maya', async (req, res) => {
  const { name, dob, intention, lang } = req.body;
  const [year, month, day] = dob.split('-').map(Number);

  });
  res.json({ ...reading, tier: hasFullAccess(req, ['maya_full']) ? 'full' : 'basic' });
});
```
