function createCheckout(btn, tier, price) {
  // 获取用户货币偏好
  const userRegion = detectUserRegion(); // 'cn'/'en'/'kr'
  const currency = getCurrency(userRegion);
  const amount = convertPrice(price, currency);
  
  // 发送到后端
  fetch('/api/create-checkout', {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${token}` },
    body: JSON.stringify({
      tier: tier,
      amount: amount,
      currency: currency,
      region: userRegion,
      referral_code: getReferralCode() // 邀请码
    })
  });
}

// 支付成功后
if (order.referral_code) {
  const referrerUser = findUserByRefCode(order.referral_code);
  if (referrerUser) {
    // 1. 计算返佣金额
    const rewardAmount = calculateReward(order.amount, order.currency);
    
    // 2. 创建返佣记录
    _M.referral_rewards.push({
      referrer_id: referrerUser.id,
      referee_id: order.user_id,
      amount: rewardAmount,
      currency: order.currency,
      status: 'pending',
      created_at: now()
    });
    
    // 3. 更新统计
    updateReferralStats(referrerUser.id);
    
    // 4. 发送通知邮件/推送
    notifyReferralReward(referrerUser, rewardAmount);

}
